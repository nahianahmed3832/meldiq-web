			81 text import { Bot } from './yapper.js';
const bot = new Bot('yb_…');                    // your token

// Commands: people type /name (or !name). They show up in the "/" menu for everyone in the server.
bot.command('ping', 'Is the bot alive?', (msg) => msg.reply('pong'));
bot.command('roll', 'Roll a die, like /roll 20', (msg) => {
  const n = Number(msg.argv[0]);                 // msg.args = "20", msg.argv = ["20"]
  const sides = Number.isSafeInteger(n) && n >= 2 && n <= 1000 ? n : 6;   // no -5, 2.5 or Infinity
  msg.reply(`🎲 ${1 + Math.floor(Math.random() * sides)} (d${sides})`);
});

bot.on('message', (msg) => {                    // every message in the bot's servers (not its own)
  if (msg.mentioned) msg.reply(`hey ${msg.authorMention}`);
});
bot.on('member_join', (ev) => bot.send(someChannelId, `welcome <@${ev.user_id}>, member #${ev.member_count}`));
bot.on('member_leave', (ev) => console.log(ev.username, 'left'));
bot.on('ready', (me) => console.log('online as', me.username));
bot.on('error', (e) => console.error(e));

bot.start();                                    // registers the commands, then listens until you stop it

			1193 heading How it works, Value: 2, ID: how-it-works
				1194 text How it works
			1195 text Yapin’ keeps every event for your bot for a day: new messages in its servers, joins, leaves, reactions, button clicks, voice joins and leaves, tests. Your bot asks
			1196 text  GET /events?after=<last id>&wait=20
			1197 text ; the call returns as soon as there is something (up to 100 events), or after about 20 seconds with an empty list. The SDK runs that loop for you. Events come in order and each id once; if a poll fails, ask again with the same 
			1198 text after
			1199 text  and nothing is lost. (A bot that was offline for a while skips slash commands older than 15 minutes and button clicks that already expired, see 
			1200 text /events
			1201 text  below.) No public URL, no port forwarding, works on a laptop behind any router. If you do host the bot somewhere, set a webhook URL in Settings → Developer → your bot → Advanced and the same events are also POSTed there (see Event webhooks). Messages people send in DMs aren't sent to bots as events. A bot can still open a DM and post there.

  const name = msg.args.replace(/[^\w-]/g, '');

  const here = voice.find((v) => v.user_ids.includes(msg.author_id));

  }

  await call.leave();

async def play(server_id, channel_id, path):
    call = await bot.connect_voice(server_id, channel_id)
    ff = await asyncio.create_subprocess_exec("ffmpeg", "-loglevel", "error", "-i", path,
        "-f", "s16le", "-ar", "48000", "-ac", "2", "pipe:1", stdout=asyncio.subprocess.PIPE)

        try:
            chunk = await ff.stdout.readexactly(FRAME)
        except asyncio.IncompleteReadError:
            break                            # end of the song
        await call.source.capture_frame(rtc.AudioFrame(chunk, 48000, 2, 480))   # waits, so it plays in real time
    await call.source.wait_for_playout()
    await call.leave()

    msg.reply(f"Playing {name}")
    threading.Thread(target=lambda: asyncio.run(play(msg.server_id, here["channel_id"], f"music/{name}.mp3")), daemon=True).start()
